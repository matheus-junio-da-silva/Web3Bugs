rule additivity_deposit() {
    uint256 x;
    uint256 y;
    env e; 
    require x > 0 && y > 0;

    storage init = lastStorage;

    deposit(e, x);
    deposit(e, y);

    uint256 bal_sep = userJoeStaked(e.msg.sender);

    uint256 bal_sum = userJoeStaked(e.msg.sender);

}
